1. Online Course Subscription System
class Course {
    private String title;
    private String duration;
    private int price;

    // constructor + getters
}

class Student {
    private String name;
    private String email;
    private List<Course> enrolledCourses;

    // constructor
    void enrollCourse(Course c) {
        if (!enrolledCourses.contains(c)) {
            enrolledCourses.add(c);
        } else {
            print("Already enrolled in: " + c.getTitle());
        }
    }

    void viewEnrolledCourses() {
        print(name + "'s Enrolled Courses:");
        for (Course c : enrolledCourses) {
            print(c.getTitle() + " - " + c.getDuration() + " - ₹" + c.getPrice());
        }
    }
}

// Driver
Course java = new Course("Java Programming Basics", "6 weeks", 4999);
Course ds = new Course("Data Structures in Java", "8 weeks", 5999);
Course spring = new Course("Spring Boot for Web Development", "10 weeks", 6999);

Student amit = new Student("Amit", "amit@email.com");
amit.enrollCourse(java);
amit.enrollCourse(ds);
amit.viewEnrolledCourses();

Student priya = new Student("Priya", "priya@email.com");
priya.enrollCourse(spring);
priya.enrollCourse(java);
priya.viewEnrolledCourses();

1. Product Search Bar (Binary Search)

String[] catalog = {"AirPods", "Backpack", "Camera", "Laptop", "Shoes", "Smartphone", "Watch"};

boolean binarySearch(String[] arr, String target) {
    int low = 0, high = arr.length - 1;
    while (low <= high) {
        int mid = (low + high) / 2;
        if (arr[mid].equals(target)) return true;
        else if (arr[mid].compareTo(target) < 0) low = mid + 1;
        else high = mid - 1;
    }
    return false;
}

// Driver
if (binarySearch(catalog, "Laptop"))
    print("Laptop available in Electronics");
else
    print("Laptop not found");

2. Payment Gateway Processor (Generics)

class Transaction<T> {
    private T payment;

    void processPayment() {
        if (payment is UPI) print("Payment via UPI...");
        else if (payment is Card) print("Payment via Card...");
        else if (payment is NetBanking) print("Payment via NetBanking...");
    }
}

class UPI { String upiId; int amount; String merchant; }
class Card { String cardNumber; String cvv; String expiry; int amount; }
class NetBanking { String bankName; String account; int amount; }

// Driver
Transaction<UPI> t1 = new Transaction<>(new UPI("ramesh@upi", 1200, "Amazon"));
t1.processPayment();

Transaction<Card> t2 = new Transaction<>(new Card("1234-5678-9876-5432", "123", "12/27", 5000));
t2.processPayment();
